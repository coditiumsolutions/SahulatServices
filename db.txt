  -- Exactly one row per provider should have IsPrimary=1, kept in sync with the deprecated
  -- Providers.CategoryUID scalar by ProviderCategoryService.SyncCategoriesAsync (the only
  -- intended writer of Providers.CategoryUID going forward).
  -- Backfilled from every existing Providers.CategoryUID at migration time (IsPrimary=1).
  -- Catch-up (2026-09-23): 6 providers created in the gap between the migration's backfill and

  -- backfill query re-run manually to close the gap. No such gap should recur going forward.
  -- Managed via GET/PUT /api/providers/{providerUid}/categories (see api.txt). Job-inbox
  -- matching, nearby-provider search, and the category-delete in-use check all read this
  -- table (Any() match against ANY of a provider's categories) rather than the scalar.
  -- Index: IX_ProviderCategories_CategoryUID; UQ_ProviderCategories_ProviderUID_CategoryUID

Table: ServiceBookings
--------------------------------------------------------------------------------
  UID                  int              NOT NULL  PK
  RequestUID           int              NOT NULL  FK -> CustomerServiceRequests.UID
  ClientUID            int              NOT NULL  FK -> Clients.UID
  ProviderUID          int              NOT NULL  FK -> Providers.UID
  ServiceDetail        nvarchar(1000)   NULL
  EstimatedAmount      decimal(12,2)    NOT NULL  DEFAULT (0)
  LabourAmount         decimal(12,2)    NULL      -- added 2026-09-21. Labour-only charge; CommissionAmount/
                                                   -- ProviderEarning are computed off THIS (not FinalAmount) once
                                                   -- set. NULL = not yet entered — legacy bookings, or a booking
                                                   -- completed by an app build that doesn't send it yet (see
                                                   -- VerifyCompletionPasscodeDto in api.txt); commission falls back
                                                   -- to the whole FinalAmount in that case (old, pre-split behavior)
  VisitCharges         decimal(12,2)    NOT NULL  DEFAULT (0)
  AdditionalCharges    decimal(12,2)    NOT NULL  DEFAULT (0)
  Deductions           decimal(12,2)    NOT NULL  DEFAULT (0)
  FinalAmount          decimal(12,2)    NOT NULL  Final Bill = Labour + Material + Estimated + Visit + Additional - Deductions
                                                   -- (Material = SUM(BookingMaterialItems.Amount) for this booking, not a stored column)
  CustomerPaid         decimal(12,2)    NOT NULL  DEFAULT (0)
  PaymentMode          nvarchar(30)     NOT NULL  CHECK (CashToProvider | OnlineToCompany)
  CustomerRemaining    decimal(12,2)    NOT NULL  DEFAULT (0)  Final Bill - Customer Paid
  CommissionType       nvarchar(10)     NOT NULL
  CommissionValue      decimal(10,2)    NOT NULL
  CommissionAmount     decimal(12,2)    NOT NULL  Company Commission
  ProviderEarning      decimal(12,2)    NOT NULL
  Status               nvarchar(20)     NOT NULL  DEFAULT ('Completed')
  -- Status CHECK: Pending | Accepted | In Progress | Completed | Closed | Cancelled | Rejected
  -- Pending = assigned by staff, awaiting provider response
  -- Rejected = provider declined; staff-only, hidden from client/provider-facing API
  Passcode             nvarchar(10)     NULL      -- 4-digit completion passcode, set when provider accepts
  AcceptedOn           datetime         NULL      -- set when provider accepts the assignment
  CompletedOn          datetime         NULL      -- set when provider verifies the passcode
  CancelReason         nvarchar(500)    NULL      -- set when Accepted/In Progress booking is cancelled by customer/provider/staff;
                                                   -- CancelReason="Assigned to another provider" specifically marks a Pending sibling
                                                   -- booking that was auto-superseded because another fanned-out provider accepted first
                                                   -- (first-accept-wins; see BookingService.RespondToAssignmentAsync)
  RejectReason         nvarchar(500)    NULL      -- set when provider rejects a Pending booking
  CreatedOn            datetime         NOT NULL  DEFAULT (getutcdate())
  -- Finance post (auto when Status=Completed):
  --   PaymentLedger: Company/Credit/Commission, Provider/Credit/JobEarning,
  --                  CustPayment (OnlineToCompany) or CashCollect (CashToProvider)
  --   ProviderPayouts: Pending Online payout when PaymentMode=OnlineToCompany
  --   CustomerPaid/PaymentMode are finalized at passcode-verification time (actual amount collected)
